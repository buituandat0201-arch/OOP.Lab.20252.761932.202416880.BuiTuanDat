SELECT 
    department_id,
    TO_CHAR(hire_date, 'MM') AS month,
    COUNT(*) AS total_employees
FROM employees
GROUP BY department_id, TO_CHAR(hire_date, 'MM')
ORDER BY department_id, month;

-- Đổi kiểu dữ liệu
ALTER TABLE locations 
MODIFY postal_code VARCHAR2(12);

-- Cập nhật dữ liệu
UPDATE locations
SET postal_code =
    CASE
        WHEN LENGTH(postal_code) < 5 
            THEN RPAD(postal_code, 5, '0')
        WHEN REGEXP_LIKE(postal_code, '[A-Za-z]') 
            THEN ASCII(SUBSTR(postal_code,1,1))
        WHEN LENGTH(postal_code) > 12 
            THEN SUBSTR(postal_code, 1, 12)
        ELSE postal_code
    END;

-- câu 9
SELECT 
    e.employee_id,
    e.first_name,
    j.job_id,
    j.start_date,
    j.end_date,
    TRUNC(MONTHS_BETWEEN(j.end_date, j.start_date)/12) AS years,
    MOD(TRUNC(MONTHS_BETWEEN(j.end_date, j.start_date)),12) AS months,
    j.end_date - ADD_MONTHS(j.start_date, TRUNC(MONTHS_BETWEEN(j.end_date, j.start_date))) AS days
FROM employees e
JOIN job_history j ON e.employee_id = j.employee_id
ORDER BY e.employee_id, j.start_date;
-- câu 10 
SELECT 
    department_id,
    job_id,
    COUNT(DISTINCT employee_id) AS total_employees
FROM (
    SELECT employee_id, department_id, job_id FROM employees
    UNION
    SELECT employee_id, department_id, job_id FROM job_history
)
GROUP BY department_id, job_id
ORDER BY department_id, job_id;
